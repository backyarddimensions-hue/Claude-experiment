+// fullyConst reports whether a query value is built only from string constants.
+func fullyConst(v ssa.Value, depth int) bool {
+	if depth > 8 {
+		return false
+	}
+	switch x := v.(type) {
+	case *ssa.Const:
+		return x.Value != nil && isString(x.Type())
+	case *ssa.BinOp:
+		return x.Op == token.ADD && fullyConst(x.X, depth+1) && fullyConst(x.Y, depth+1)
+	case *ssa.UnOp:
+		if al, ok := x.X.(*ssa.Alloc); ok && x.Op == token.MUL {
+			n := 0
+			okAll := true
+			for _, r := range *al.Referrers() {
+				if st, ok := r.(*ssa.Store); ok {
+					n++
+					okAll = okAll && fullyConst(st.Val, depth+1)
+				}
+			}
+			return n > 0 && okAll
+		}
+	}
+	return false
+}
